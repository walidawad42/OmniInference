cmake_minimum_required(VERSION 3.20)
project(OmniInference VERSION 1.0.0 LANGUAGES C CXX CUDA)

# --- 1. DETERMINISTIC BUILD CONFIGURATION ---
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# HARDWARE TARGETING: NVIDIA Quadro K5100M (Kepler GK110)
set(CMAKE_CUDA_ARCHITECTURES 30 CACHE STRING "CUDA architectures" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

# --- 2. EXTERNAL DEPENDENCIES (FetchContent) ---
include(FetchContent)

# 1. Fetch TurboQuant Plus from GitHub
message(STATUS "Fetching TurboQuant Plus...")
FetchContent_Declare(
    TurboQuantPlus
    GIT_REPOSITORY https://github.com/TheTom/turboquant_plus.git
    GIT_TAG main
)
FetchContent_MakeAvailable(TurboQuantPlus)

# 2. Fetch/Find llama.cpp (GGML Backend)
# Assuming you still want llama.cpp for the inference engine
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/llama.cpp/CMakeLists.txt")
    set(GGML_CUDA ON CACHE BOOL "" FORCE)
    set(GGML_VULKAN ON CACHE BOOL "" FORCE)
    add_subdirectory(vendor/llama.cpp)
else()
    message(FATAL_ERROR "llama.cpp backend not found in vendor/")
endif()

# Find standard libraries
find_package(CUDAToolkit 11.8 REQUIRED)
find_package(Vulkan REQUIRED)
find_package(SDL2 REQUIRED)

# --- 3. APPLICATION EXECUTABLE ---
add_executable(OmniInference 
    "src/main.cpp"
    "src/omni_engine.cpp"
    "src/gui_app.cpp"
    "src/server.cpp"
    "src/markdown_exporter.cpp"
)

target_include_directories(OmniInference PRIVATE 
    ${Vulkan_INCLUDE_DIRS}
    ${SDL2_INCLUDE_DIRS}
    "${CMAKE_CURRENT_SOURCE_DIR}/vendor/llama.cpp"
    
    # Include the fetched TurboQuant Plus headers
    ${turboquant_plus_SOURCE_DIR}/include
)

# Compile Definitions
target_compile_definitions(OmniInference PRIVATE
    TURBOQUANT_PLUS_ENABLED
    ROTORQUANT_BLOCKED=1
)

# --- 4. LINKING ---
target_link_libraries(OmniInference PRIVATE 
    llama
    ggml
    
    # Link the fetched TurboQuant Plus library target
    # Note: Check the repo's CMakeLists.txt for the exact target name. 
    # It is usually 'turboquant_plus' or similar.
    turboquant_plus 

    ${Vulkan_LIBRARIES}
    SDL2::SDL2main
    SDL2::SDL2
    CUDA::cudart
    CUDA::cublas
    d3d11 dxgi d3dcompiler
)

# --- 5. INSTALL & BUNDLE ---
install(TARGETS OmniInference RUNTIME DESTINATION bin)

# Bundle CUDA 11.8 Runtime (Critical for sm_30)
function(bundle_dependency NAME DLL_PATH)
    if(EXISTS "${DLL_PATH}")
        install(FILES "${DLL_PATH}" DESTINATION bin)
    endif()
endfunction()

bundle_dependency("CUDA Runtime" "${CUDAToolkit_BIN_DIR}/cudart64_110.dll")
bundle_dependency("cuBLAS" "${CUDAToolkit_BIN_DIR}/cublas64_11.dll")
bundle_dependency("cuBLASLt" "${CUDAToolkit_BIN_DIR}/cublasLt64_11.dll")

# --- 6. CPACK CONFIGURATION ---
set(CPACK_PACKAGE_NAME "OmniInference")
set(CPACK_PACKAGE_VENDOR "Omni-Inference Project")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_INSTALL_DIRECTORY "OmniInference")
set(CPACK_GENERATOR "WIX")

set(CPACK_WIX_UPGRADE_GUID "A9B8C7D6-E5F4-3A2B-1C0D-9E8F7A6B5C4D")
set(CPACK_WIX_PRODUCT_GUID "B1C2D3E4-F5A6-7890-ABCD-EF1234567890")

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE.rtf")
    set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE.rtf")
endif()

set(CPACK_CREATE_DESKTOP_LINKS "OmniInference")
include(CPack)
